
from collections import deque

graph = {}

n = int(input("Enter number of nodes: "))

# Taking nodes
for i in range(n):
    node = input("Enter node: ")
    graph[node] = []

e = int(input("Enter number of edges: "))

# Taking edges
for i in range(e):
    u = input("Enter first node: ")
    v = input("Enter second node: ")

    if u in graph and v in graph:
        graph[u].append(v)
        graph[v].append(u)
    else:
        print("Invalid edge!")

start = input("Enter starting node: ")
goal = input("Enter goal node: ")

# BFS
def bfs(graph, start, goal):
    queue = deque([[start]])
    visited = {start}

    while queue:
        path = queue.popleft()
        node = path[-1]

        if node == goal:
            return path

        for neighbour in graph[node]:
            if neighbour not in visited:
                visited.add(neighbour)
                queue.append(path + [neighbour])

    return None

# DFS
def dfs(graph, start, goal):
    stack = [[start]]
    visited = set()

    while stack:
        path = stack.pop()
        node = path[-1]

        if node == goal:
            return path

        if node not in visited:
            visited.add(node)

            for neighbour in graph[node]:
                if neighbour not in visited:
                    stack.append(path + [neighbour])

    return None

# Calling functions
if start in graph and goal in graph:
    bfs_path = bfs(graph, start, goal)
    dfs_path = dfs(graph, start, goal)

    print("\nBFS Path:", bfs_path)
    print("DFS Path:", dfs_path)

    if bfs_path:
        print("BFS Path Length:", len(bfs_path) - 1)

    if dfs_path:
        print("DFS Path Length:", len(dfs_path) - 1)
else:
    print("Invalid starting or goal node!")


1. Data structures: BFS uses a queue (FIFO), DFS uses a stack (LIFO) or recursion. Both use a visited
set.
2. Complexity: BFS time O(b^d), space O(b^d). DFS time O(b^m), space O(b·m), where d = goal depth
and m = max depth.
3. BFS vs DFS: BFS is complete and gives the shortest path (unweighted), but uses more memory. DFS
uses less memory but may give a longer path and is not optimal.
4. Limited memory: Choose DFS, because it only stores the current path.
5. Performance comparison: BFS gives the shortest path length but usually explores more nodes. DFS
may explore fewer or more nodes and gives a path that is not necessarily shortest.
