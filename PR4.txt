
# Forward and Backward Chaining

fever = input("Do you have fever? (yes/no): ").lower()
cough = input("Do you have cough? (yes/no): ").lower()

# Initial facts
facts = set()

if fever == "yes":
    facts.add("fever")

if cough == "yes":
    facts.add("cough")

# Rules
rules = [
    (["fever", "cough"], "flu"),
    (["flu"], "take_rest"),
    (["fever"], "drink_water")
]


# Forward Chaining
def forward_chaining(facts, rules):
    facts = set(facts)
    changed = True

    while changed:
        changed = False

        for conditions, result in rules:
            if result not in facts and all(c in facts for c in conditions):
                facts.add(result)

                print("Rule fired:",
                      " AND ".join(conditions), "->", result)

                changed = True

    return facts


# Backward Chaining
def backward_chaining(goal, facts, rules, depth=0):
    indent = " " * depth

    if goal in facts:
        print(indent, goal, "is a known fact")
        return True

    for conditions, result in rules:
        if result == goal:
            print(indent, "To prove", goal,
                  "we need:", conditions)

            if all(backward_chaining(c, facts, rules, depth + 1)
                   for c in conditions):
                return True

    return False


# Display results
print("\nInitial Facts:", facts)

print("\n--- Forward Chaining ---")
final_facts = forward_chaining(facts, rules)
print("Final Facts:", final_facts)

print("\n--- Backward Chaining ---")
goal = input("Enter goal (flu/take_rest/drink_water): ")

if backward_chaining(goal, facts, rules):
    print("Goal", goal, "is proved.")
else:
    print("Goal", goal, "cannot be proved.")


1. Expert system: A program that imitates a human expert's decision-making in a specific domain.
Components: knowledge base, inference engine, user interface, working memory (facts), and an
explanation facility (plus a knowledge acquisition module).
2. Fact vs rule: A fact is a known true statement (fever). A rule is an IF-THEN relation (IF fever AND cough
THEN flu).
3. Propositional logic: Statements that are either true or false, combined with AND, OR, NOT, and
IMPLIES. It is used to represent rules and facts in the knowledge base.
4. Forward vs backward chaining: Forward is data-driven: it starts from facts and applies rules to
derive new facts until the goal appears. Backward is goal-driven: it starts from the goal and checks
whether the premises can be proved.
5. Role of knowledge base: It stores domain facts and rules, and the inference engine uses it to reach
conclusions.
6. How backward chaining proves a goal: It looks for rules that conclude the goal, then recursively
tries to prove each premise, until all reduce to known facts.
